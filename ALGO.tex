\subsection*{Lineární funkce}
\begin{definition}
Lineární funkce má tvar
\[
f(x) = ax + b, \quad a \neq 0.
\]
\end{definition}

\textbf{Vlastnosti:}
\begin{itemize}
  \item definiční obor: $\mathbb{R}$
  \item obor hodnot: $\mathbb{R}$
  \item monotónnost: rostoucí pro $a>0$, klesající pro $a<0$
  \item nemá extrémy, není omezená
\end{itemize}

\chapter{Posloupnosti}
\begin{definition}
\emph{Posloupnost} je zobrazení z množiny přirozených čísel do libovolné množiny.
Zápis: $(a_n)$ -- posloupnost, $a_n$ -- n-tý prvek.
\end{definition}

Často za množinu $A$ budeme dosazovat $\mathbb{R}$, takovým posloupnostem se říká \emph{číselné posloupnosti}. Ty budou důležité pro tuto kapitolu. O posloupnosti $(a_n):\mathbb{N} \rightarrow A$ můžeme uvažovat tak, že každému prvku z $A$ lze přiřadit index.

\section{Způsoby zadávání posloupností} 
	\subsection{Prvních x členů}
		Musí být jasné pravidlo, jak vyjádřit další členy posloupnosti.
		\begin{example}
			\begin{itemize}
				\item triviální příklad $1, 2, 3, 4 ... $
				\item fibbonacci $0, 1, 1, 2, 3, 5, 8 ...$
				\item alternující $1, -1, 1, -1 ...$
				\item konečná $2, 4, 6, ..., 20$
			\end{itemize}
		\end{example}
	\subsection{Vzorec pro n-tý člen}
		Vyjádříme obecný vzorec pro n-tý prvek na základě indexu.
		\begin{example}
			\begin{itemize}
				\item $(\frac{n}{n+1})$
				\item $((-1)^nn)$
				\item $ (1 + \frac{1}{n})^n)$
			\end{itemize}
		\end{example}
	
	\subsection{Rekurentně} 
		Rekurentní zadání obsahuje zpravidla 1. člen (nebo několik prvních členů) a pravidlo, jak vytvořit další člen ze členů předcházejících.
		\begin{example}
			\begin{itemize}
				\item $(\frac{n}{n+1})$
				\item $((-1)^nn)$
				\item $ (1 + \frac{1}{n})^n)$
			\end{itemize}
		\end{example}
Speciálním případem rekurentního zadání jsou aritmetická a geometrická posloupnost.

\begin{definition}
	Posloupnost $(a_n)$ se nazývá \textbf{aritmetická} právě tehdy, když
	\[
	\exists d \in \mathbb{R} \ \forall n \in \mathbb{N} \quad a_{n+1} = a_n + d
	\]
	Číslo $d$ se nazývá \textbf{diference} aritmetické posloupnosti.
\end{definition}

\begin{definition}
	Posloupnost $(a_n)$ se nazývá \textbf{geometrická} právě tehdy, když
	\[
	\exists q \in \mathbb{R} \ \forall n \in \mathbb{N} \quad a_{n+1} = a_n \cdot q
	\]
	Číslo $q$ se nazývá \textbf{kvocient} aritmetické posloupnosti.
\end{definition}
		
\section{Vlastnosti posloupností}
\begin{definition}
Posloupnost $(a_n)$ nazveme
\begin{itemize}
  \item \emph{rostoucí}, jestliže $a_{n+1} > a_n$ pro všechna $n \in \mathbb{N}$,
  \item \emph{nerostoucí}, jestliže $a_{n+1} \leq a_n$ pro všechna $n \in \mathbb{N}$,
  \item \emph{klesající}, jestliže $a_{n+1} < a_n$ pro všechna $n \in \mathbb{N}$,
  \item \emph{neklesající}, jestliže $a_{n+1} \geq a_n$ pro všechna $n \in \mathbb{N}$.
\end{itemize}
\end{definition}

\begin{definition}
Posloupnost $\{a_n\}$ je \emph{omezená}, jestliže existuje reálné číslo $M > 0$ tak, že
\[
  |a_n| \leq M \quad \text{pro všechna } n \in \mathbb{N}.
\]
Ekvivalentně, posloupnost je omezená shora, pokud $\exists K \in \mathbb{R}$ tak, že $a_n \leq K$ pro všechna $n$;  
omezená zdola, pokud $\exists L \in \mathbb{R}$ tak, že $a_n \geq L$ pro všechna $n$.
\end{definition}


\begin{definition}
Pro posloupnost $\{a_n\}$ definujeme:
\begin{itemize}
  \item \emph{maximum} $\max\{a_n\}$, pokud existuje člen $a_k$, pro který platí $a_k \geq a_n$ pro všechna $n$,
  \item \emph{minimum} $\min\{a_n\}$, pokud existuje člen $a_k$, pro který platí $a_k \leq a_n$ pro všechna $n$.
\end{itemize}
\end{definition}

\section{Základní otázky o algoritmech}

\subsection{Správnost}
Vrátí algoritmus správný výsledek pro každý přípustný vstup?

\begin{algorithm}
\caption{Algoritmus pro nalezení maxima pole}
\begin{algorithmic}[1]
\Require Pole $A$ délky $n$ obsahující celá čísla
\Ensure Maximální prvek v poli
\State $max \gets 0$
\For{$i = 1$ to $n-1$}
    \If{$A[i] > max$}
        \State $max \gets A[i]$
    \EndIf
\EndFor
\State \Return max
\end{algorithmic}
\end{algorithm}

Původní verze byla nesprávná, protože počáteční hodnota \texttt{max} byla nastavena na 0.  
Pro pole se zápornými hodnotami tak algoritmus vracel chybný výsledek.

---

\begin{algorithm}
\caption{Lineární vyhledávání}
\begin{algorithmic}[1]
\Require Pole $A$ délky $n$, hledaná hodnota $x$
\Ensure Index prvku $x$ nebo \textsc{NIL}, pokud se nenachází
\For{$i = 0$ to $n - 1$}
    \If{$A[i] = x$}
        \State \Return $i$
    \EndIf
\EndFor
\State \Return \textsc{NIL}
\end{algorithmic}
\end{algorithm}

\begin{proof}
Chceme dokázat, že algoritmus vrátí správný výsledek a že vždy skončí.

Na začátku každé iterace s indexem $i$ platí indukční předpoklad:  
„Prvek $x$ není v $A[0\,..\,i-1]$.“

\begin{itemize}
  \item Pro $i=0$ je tvrzení triviálně pravdivé.
  \item Pokud algoritmus v iteraci $i$ zjistí, že $A[i] = x$, vrátí správně index.
  \item Pokud $A[i] \neq x$, je tvrzení splněno i pro další iteraci.
\end{itemize}

Cyklus skončí po $n$ iteracích. Pokud se $x$ v poli nenachází, algoritmus vrátí \textsc{NIL}, což je správně.
\end{proof}

\subsubsection{Časová složitost}

Popisuje, jak rychle algoritmus pracuje – tedy závislost počtu vykonaných kroků na velikosti vstupu.  
Tuto závislost můžeme vyjádřit funkcí
\[
T: \mathbb{N} \to \mathbb{N},
\]
která velikosti vstupu přiřadí počet kroků provedených algoritmem.

Složitosti se budeme podrobně věnovat v další kapitole.

\subsection{Optimalita}
Je náš algoritmus nejlepší možný vzhledem ke složitosti?  
Neexistuje rychlejší varianta?

\begin{algorithm}
\caption{Největší společný dělitel — naivní verze}
\begin{algorithmic}
\Require Celá čísla $m, n$
\Ensure Největší společný dělitel čísel $m, n$
\State $t \gets \min(m, n)$
\While{$m \bmod t \neq 0$ \textbf{ or } $n \bmod t \neq 0$}
	\State $t \gets t - 1$
\EndWhile
\State \Return t
\end{algorithmic}
\end{algorithm}

Správnost je zřejmá, ale algoritmus je velmi pomalý.  
Existuje mnohem rychlejší varianta – \textbf{Eukleidův algoritmus}.

\chapter{Složitost}

Analýza algoritmu - snažíme se odhadnout jak se algoritmus bude chovat. 
Pokud mluvíme o složitosti jde nám o to kolik algoritmu zabere času než nám dá výsledek.
Nebo můžeme uvažovat paměťovou složitost -- kolik paměti algoritmus bude potřebovat na výpočet.

pokud budeme dále mluvit o složitosti algoritmu. Budeme tím myslet časovou, pokud nebude explicitně řečeno jinak. 

\begin{definition}
	\emph{Časová složitost algoritmu} $A$ je funkce $T_A: \mathbb{N} \rightarrow \mathbb{N}$
	$T_A(n) =$ počet elementárních kroků, které $A$ vykoná pro vstup velikosti n.
\end{definition}

Funkce $T_A(n)$ popisuje jak se algoritmus chová pro všechny vstupy.
Obvykle nás zajímá nejhorší případ, nebo průměrný případ.


Nechť algoritmus $A$ řeší problém $P$ a nechť $I_1, I_2, \dots, I_m$ jsou všechny možné vstupy problému $P$ o velikosti $n$.  
Označme $t_A(I_i)$ dobu běhu algoritmu $A$ na instanci $I_i$.

\begin{definition}
\textbf{Časová složitost v nejhorším případě} (worst-case time complexity) je definována jako maximální doba běhu algoritmu pro všechny možné instance o velikosti $n$:
\[
T_A^{\max}(n) = max\{t_A(I)|\text{I je vstup problému P a} |I| = n\}
\]

\[
\qquad \text{tj. graficky:} \qquad
\left.
\begin{array}{c|c}
I_1 & t_A(I_1) \\
I_2 & t_A(I_2) \\
\vdots & \vdots \\
I_m & t_A(I_m)
\end{array}
\right\}
\text{max.}
\]
\end{definition}
\newpage
\begin{definition}
\textbf{Časová složitost v průměrném případě} (average-case time complexity) je definována jako průměrná doba běhu algoritmu:
\[
T_A^{\text{avg}}(n) = \frac{t_A(I_1) + \dots + t_A(I_m)}{m}
\qquad \text{tj. graficky:} \qquad
\left.
\begin{array}{c|c}
I_1 & t_A(I_1) \\
I_2 & t_A(I_2) \\
\vdots & \vdots \\
I_m & t_A(I_m)
\end{array}
\right\}
\mathbb{E}
\]
kde $p_i$ je pravděpodobnost výskytu instance $I_i$.
\end{definition}

\begin{definition}
\emph{Problém třídění:}

\textbf{Vstup:} posloupnost n čísel: $\langle a_1, a_2 \dots a_n \rangle$

\textbf{Výstup:} permutace $\langle b_1, b_2 \dots b_n \rangle$ tak, že platí 
$b_1 \leq b_2 \leq \dots \leq b_n$

vstup i výstup obvykle reprezentujeme polem.
\end{definition}

\begin{definition}

Algoritmus pracuje \emph{na místě} pokud použije nanejvýš konstantní prostor mimo tříděné pole.  

\end{definition}


\section{Insertion Sort}

Idea tohoto algoritmu je podobná způsobu, jak třídíme \( n \) rozdaných karet:  
\( n \) karet leží na začátku na stole. Pravou rukou je bereme a vkládáme do levé ruky tak, že v levé ruce vzniká setříděná posloupnost karet (zleva od nejmenší po největší). Drží-li levá ruka \( k \) karet, pak další, \((k + 1)\)-ní, kartu zatřídíme tak, že ji zprava porovnáváme se setříděnými kartami a vložíme ji na správné místo.

\begin{algorithm}[h!]
\caption{Insertion-Sort(\(A[0..n-1]\))}
\begin{algorithmic}[1]
\For{$j \gets 1$ \textbf{to} $n - 1$}
  \State $t \gets A[j]$
  \State $i \gets j - 1$
  \While{$i \ge 0$ \textbf{and} $A[i] > t$}
    \State $A[i + 1] \gets A[i]$
    \State $i \gets i - 1$
  \EndWhile
  \State $A[i + 1] \gets t$
\EndFor
\end{algorithmic}
\end{algorithm}

\section{Selection Sort}

Idea: Najdi v \( A[0..n-1] \) nejmenší prvek a vyměň ho s \( A[0] \).  
Najdi v \( A[1..n-1] \) nejmenší prvek a vyměň ho s \( A[1] \).  
\(\ldots\)  
Najdi v \( A[n-2..n-1] \) nejmenší prvek a vyměň ho s \( A[n-2] \).

\begin{algorithm}[h!]
\caption{Selection-Sort(\(A[0..n-1]\))}
\begin{algorithmic}[1]
\For{$j \gets 0$ \textbf{to} $n - 2$}
  \State $iMin \gets j$
  \For{$i \gets j + 1$ \textbf{to} $n - 1$}
    \If{$A[i] < A[iMin]$}
      \State $iMin \gets i$
    \EndIf
  \EndFor
  \State $t \gets A[j]; \; A[j] \gets A[iMin]; \; A[iMin] \gets t$
\EndFor
\end{algorithmic}
\end{algorithm}

		\item Naprogramujte insertion sort a selection sort v pythonu. 
		\item Upravte selection sort tak, aby fungoval výběrem největšího prvku.
		\item Upravte insertion sort tak, aby třídíl sestupně nebo vzestupně na základě volitelného parametru.
	\end{enumerate}


\chapter{Asymptotický růst funkcí}

V předchožích kapitolách jsme počítaly kolik kroků algoritmus vykoná v nejhorším případě (časová složistos). V této kapitole se budeme věnovat časové složitosti více do hloubky. Zavedeme $O$-notaci.
Ukážeme jak analyzovat algoritmy efektivněji bez zbytečně velké přesnosti, která nám nedává informaci navíc. 
Dejme tomu, že algoritmus $A$ a $B$ řeší problém $P$. Spočítáním počtu instrukcí co provedou máme: 
$T_A(n) = 2n + 4$ a $T_B(n) = 6n - 10$. 

Vidíme, že funkce $T_A(n)$ roste rychleji než $T_B(n)$ rozdíl je ale nezávislý na n.
Podívejme se tedy na hrubší metodu analýzy, která tento nedostatek odstraní.

\begin{definition}
\emph{Asymptotická hodní mez $O(g)$} pro funkci $g(n)$ je 
\[
O(g(n)) = \{f(n) \ | \ \exists c > 0 \ \text{a} \ n_0 \in \mathbb{N} \text{ tak, že } \forall n \geq n_0 \text{ je } 0 \leq f(n) \leq cg(n))\}
\]
\end{definition}

Všimněte, že definujeme množinu funkcí, které splňují podmínku "$f$ funkce je zhora omezená funkcí $g$".
Pak můžeme říct $n^3 + 2n + 4 \in O(n^3)$. Tedy že funkce$n^3 + 2n + 4$ roste nanejvýš tak rychle jako 
$c(n^3)$. Bude se nám hodit ještě pár definic.
U časových složitostí se někdy zapisuje $f(n) = O(g(n))$ namísto $f(n) \in O(g(n))$. Význam je ale stejný.


\begin{example}
Dokažme $2n^2 - 4 = O(n^2)$.
$f(n) = 2n^2 - 4$ a $g(n) = n^2$.
Podle definice hledáme $C$ a $n_0$
Zvolme tedy $c = 2$, $n_0 = 2$.

Pro každé $n \ge n_0$ platí $0 \le f(n) \le c g(n)$.

Tato podmínka totiž znamená, že pro každé $n \ge 2$ je 
$$0 \le 2n^2 - 4 \le 2n^2,$$
Což zřejmě platí. 

A tedy i $2n^2 - 4 = O(n^2)$ platí.
\end{example}

\newpage
\begin{example}
Dokažme, že neplatí $0.5n^3 = O(20n^2)$.

Je tedy $f(n) = 0.5n^3$ a $g(n) = 20n^2$.

Zvolme libovolné $c > 0$.

Požadovaná nerovnost pak je $0.5n^3 \le 20cn^2$, což je ekvivalentní
$$0.5n \le 20c, \text{ tj. } n \le 40c.$$

Nerovnost tedy neplatí pro žádné $n \ge 40c$. 
Neexistuje tedy $n_0$ tak, aby požadovaná nerovnost platila pro každé $n \ge n_0$.

Čísla $c$ a $n_0$ požadovaná definicí tedy neexistují, proto 
$$0.5n^3 = O(20n^2) \text{ neplatí.}$$
\end{example}

\begin{definition}
\emph{Asymptotická dolní mez $\Omega(g)$} Pro funkci $g(n)$ je
\[
\Omega(g(n)) = \{ f(n) \ | \ \exists c > 0,\ n_0 \in \mathbb{N} \forall n \ge n_0 \text{ je } 0 \le c g(n) \le f(n) \}.
\]
\end{definition}

\begin{exercise}
1) Dokažme $n^2 = \Omega(2n^2 - 4)$.
Je tedy $f(n) = n^2$ a $g(n) = 2n^2 - 4$. Zvolme $c = 1/2$ a $n_0 = 1$. 
Pak pro $n \ge n_0$ platí $cg(n) = n^2 - 2 \le n^2$. Důkaz je hotov.

\medskip

2) Dokažme $2n^2 - 4 = \Omega(n^2)$.
Je tedy $f(n) = 2n^2 - 4$ a $g(n) = n^2$. Zvolme $c = 1$ a $n_0 = 3$. 
Pak pro $n \ge n_0$ platí $cg(n) = n^2 \le 2n^2 - 4$. Důkaz je hotov.

\medskip

3) Dokažme, že pro každou funkci $f(n)$ a libovolnou $k > 0$ je
$f(n) = \Omega(k f(n))$.
Zvolme $c = 1/k$ a $n_0 = 1$. Podmínky definice pak zřejmě platí.

\end{exercise}
\begin{definition}
\emph{Asymptotická oboustranná (těsná) mez} Pro funkci $g(n)$ je
\[
\Theta(g(n)) = \{ f(n) \mid \exists\, c_1, c_2 > 0,\ n_0 \in \mathbb{N} \forall n \ge n_0 \text{ je } 0 \le c_1 g(n) \le f(n) \le c_2 g(n) \}.
\]

Alternativně platí, že $f(n) \in \Theta(g(n))$ právě když $f(n) \in O(g(n))$ a $f(n) \in \Omega(g(n))$
\end{definition}

\begin{pseudocodecardorange}{Tranzitivita odhadů}
Pokud $f = O(g)$ a $g = O(h)$, pak $f = O(h)$.\\
Pokud $f = \Omega(g)$ a $g = \Omega(h)$, pak $f = \Omega(h)$.\\
Pokud $f = \Theta(g)$ a $g = \Theta(h)$, pak $f = \Theta(h)$.\\
Pokud $f = o(g)$ a $g = o(h)$, pak $f = o(h)$.\\
Pokud $f = \omega(g)$ a $g = \omega(h)$, pak $f = \omega(h)$.
\end{pseudocodecardorange}

\begin{pseudocodecardorange}{Reflexitita a symetrie odhadů}
Reflexivita:\\
$f = O(f).$\\
$f = \Omega(f).$\\
$f = \Theta(f).$\\
Symetrie:\\
$f = \Theta(g)$ právě když $g = \Theta(f)$.\\
$f = O(g)$ právě když $g = \Omega(f)$.\\
$f = o(g)$ právě když $g = \omega(f)$.\\
\end{pseudocodecardorange}

\caption{Bubble Sort(A)}
\begin{algorithmic}[1]
    \For{$j \gets 1$ to $n-1$}
        \For{$i \gets n - 1$ downto j + 1}
            \If{$A[i] < A[i - 1]$}

            \EndIf
        \EndFor
    \EndFor
\end{algorithmic}
\end{algorithm}


\section{Quick Sort}
\emph{Intuice}: namísto řešení složitého problému najednou jej rozdělíme na menší
(jednodušší) části, které vyřešíme samostatně a jejich výsledky spojíme. Této technice se říká 
rozděl a panuj. Přímo vybízí k rekurzívní implementaci. 
\begin{enumerate}
	\item \textbf{Rozděl}: vstupní pole a rozdělíme na dvě poloviny. Dělící index se nazývá pivot.
	\item \textbf{Panuj}: rekurzivně volej quicksort na menší části pole. 
\end{enumerate}

\begin{algorithm}[H]
\caption{Quick-Sort(A, p, r)}
\begin{algorithmic}[1]
\Procedure{Quick-Sort}{A, p, r}
    \If{$p < r$}
        \State $q \gets$ \Call{Partition}{A, p, r}
        \State \Call{Quick-Sort}{A, p, q - 1}
        \State \Call{Quick-Sort}{A, q + 1, r}
    \EndIf
\EndProcedure

\Procedure{Partition}{A, p, r}
    \State $x \gets A[r]$ \Comment{pivot}
    \State $i \gets p - 1$
    \For{$j \gets p$ \textbf{to} $r - 1$}
        \If{$A[j] \le x$}
            \State $i \gets i + 1$
            \State swap($A[i]$, $A[j]$)
        \EndIf
    \EndFor
    \State swap($A[i+1]$, $A[r]$)
    \State \Return $i + 1$
\EndProcedure
\end{algorithmic}
\end{algorithm}


\section{Složitost quicksortu}
\section*{Nejhorší případ:}
\begin{itemize}
    \item nastává tehdy, když volba pivotu rozdělí pole **nerovnoměrně**
    \item pivot je vždy **nejmenší nebo největší prvek** v dané části pole
    \item v takovém případě má jedno podpole délku $n-1$ a druhé délku $0$
    \item počet instrukcí roste kvadraticky s velikostí pole, což vede na časovou složitost $O(n^2)$
\end{itemize}

\subsection*{Počty porovnání v partition (nejčastěji vykonávaná instrukce):}
\begin{enumerate}
    \item volání: $n-1$ porovnání (všechny prvky s pivotem)
    \item volání: $n-2$ porovnání
    \item volání: $n-3$ porovnání
    \item \dots
    \item (poslední) $1$ porovnání
\end{enumerate}

\medskip

\textbf{Celkem:}
$$ 1 + 2 + 3 + \dots + (n-1) = \frac{n(n-1)}{2} = \Theta(n^2) $$

\section*{Časová složitost v nejlepším a průměrném případě}

\textbf{Intuice:} Quick Sort je velmi efektivní, pokud pivoty nejsou voleny extrémně nevhodně.

\subsection*{Nejlepší případ:}
\begin{itemize}
    \item nastává, když pivot rozdělí pole na dvě **přibližně stejně velké** části
    \item každé rekurzivní volání tedy pracuje s polovinou prvků
    \item funkce partition provede $O(n)$ operací pro každou úroveň rekurze
    \item hloubka rekurze je $\log_2 n$, proto dostáváme výslednou složitost $\Theta(n \log n)$
\end{itemize}

\subsection*{Průměrný případ:}
\begin{itemize}
    \item pivoty obvykle nedělí pole zcela rovnoměrně, ale přesto rozdělí pole \glqq dostatečně dobře\grqq
    \item strom rekurzivních volání má podobnou hloubku jako v nejlepším případě
    \item proto i složitost v průměrném je lineárně-logaritmická $\Theta(n \log n)$
\end{itemize}


\section{Merge Sort}
\begin{algorithm}[H]
\caption{Merge-Sort}
\begin{algorithmic}[1]
\Procedure{Merge-Sort}{A, p, r}
    \If{$p < r$}
        \State $q \gets \lfloor (p + r)/2 \rfloor$
        \State \Call{Merge-Sort}{A, p, q}
        \State \Call{Merge-Sort}{A, q+1, r}
        \State \Call{Merge}{A, p, q, r}
    \EndIf
\EndProcedure

\Procedure{Merge}{A, p, q, r}
    \State create arrays $L$ and $R$ from $A[p \dots q]$ and $A[q+1 \dots r]$
    \State append sentinels to $L$ and $R$
    \State $i \gets 1$; $j \gets 1$
    \For{$k \gets p$ \textbf{to} $r$}
        \If{$L[i] \le R[j]$}
            \State $A[k] \gets L[i]$
            \State $i \gets i + 1$
        \Else
            \State $A[k] \gets R[j]$
            \State $j \gets j + 1$
        \EndIf
    \EndFor
\EndProcedure
\end{algorithmic}
\end{algorithm}

\section{Úkoly}
\begin{enumerate}
    \item Implementujte Bubble Sort.
    \item Upravte Bubble tak, aby používal tzv. Cocktail Shaker sort. (známí z přednášky)
          Tedy každý sudý průchod skenuje zleva doprava a každý lichý zprava doleva.
    \item Upravte Bubble sort tak, aby měl na vstupu interval $[i, j]$, kde $i$ a $j$ jsou indexy vstupního pole.
          Algoritmus by pak měl třídit pouze tuto část pole.
    \item Implementujte Quick Sort.
    \item Upravte Quick Sort tak, aby používal Hybridní třízení, tedy aby počítal rekurzivní volání a pokud překročí hranici $MaxDepth$,
          přejde v dané části pole na nějaký jednodužší sort(select, bubble, insert).
    \item Upravte Quick Sort tak, aby používal 3-cestný partition. Pole rozdělí na 3 části, prvky menší, než pivot, rovny pivotu a větší než pivot. 
    \item Implementujte Merge Sort.
    \item Upravte Merge Sort na tzv. $k$-cestný merge sort. Algoritmus na vstupu dostane číslo $k$.
          Vstupní pole pak bude dělit na $k$ částí.
          (Vyzkoušejte nejprve pro $k=3$, $k=4$, a poté pro obecné $k$.)
\end{enumerate}

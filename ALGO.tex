\subsection*{Lineární funkce}
\begin{definition}
Lineární funkce má tvar
\[
f(x) = ax + b, \quad a \neq 0.
\]
\end{definition}

\begin{itemize}
  \item Definiční obor: $\mathbb{R}$

  \item Monotónnost: rostoucí pro $a>0$, klesající pro $a<0$
  \item Nemá extrémy, není omezená
\end{itemize}

\chapter{Posloupnosti}
\begin{definition}
\emph{Posloupnost} je zobrazení z množiny přirozených čísel do libovolné množiny.
Zápis: $(a_n)$ -- posloupnost, $a_n$ -- n-tý prvek.
\end{definition}

Často za množinu $A$ budeme dosazovat $\mathbb{R}$, takovým posloupnostem se říká \emph{číselné posloupnosti}. Ty budou důležité pro tuto kapitolu. O posloupnosti $(a_n):\mathbb{N} \rightarrow A$ můžeme uvažovat tak, že každému prvku z $A$ lze přiřadit index.

\section{Způsoby zadávání posloupností} 
	\subsection{Prvních x členů}
		Musí být jasné pravidlo, jak vyjádřit další členy posloupnosti.
		\begin{example}
			\begin{itemize}
				\item triviální příklad $1, 2, 3, 4 ... $
				\item fibbonacci $0, 1, 1, 2, 3, 5, 8 ...$
				\item alternující $1, -1, 1, -1 ...$
				\item konečná $2, 4, 6, ..., 20$
			\end{itemize}
		\end{example}
	\subsection{Vzorec pro n-tý člen}
		Vyjádříme obecný vzorec pro n-tý prvek na základě indexu.
		\begin{example}
			\begin{itemize}
				\item $(\frac{n}{n+1})$
				\item $((-1)^nn)$
				\item $ (1 + \frac{1}{n})^n)$
			\end{itemize}
		\end{example}
	
	\subsection{Rekurentně} 
		Rekurentní zadání obsahuje zpravidla 1. člen (nebo několik prvních členů) a pravidlo, jak vytvořit další člen ze členů předcházejících.
		\begin{example}
			\begin{itemize}
				\item $(\frac{n}{n+1})$
				\item $((-1)^nn)$
				\item $ (1 + \frac{1}{n})^n)$
			\end{itemize}
		\end{example}
Speciálním případem rekurentního zadání jsou aritmetická a geometrická posloupnost.

\begin{definition}
	Posloupnost $(a_n)$ se nazývá \textbf{aritmetická} právě tehdy, když
	\[
	\exists d \in \mathbb{R} \ \forall n \in \mathbb{N} \quad a_{n+1} = a_n + d
	\]
	Číslo $d$ se nazývá \textbf{diference} aritmetické posloupnosti.
\end{definition}

\begin{definition}
	Posloupnost $(a_n)$ se nazývá \textbf{geometrická} právě tehdy, když
	\[
	\exists q \in \mathbb{R} \ \forall n \in \mathbb{N} \quad a_{n+1} = a_n \cdot q
	\]
	Číslo $q$ se nazývá \textbf{kvocient} aritmetické posloupnosti.
\end{definition}
		
\section{Vlastnosti posloupností}
\begin{definition}
Posloupnost $(a_n)$ nazveme
\begin{itemize}
  \item \emph{rostoucí}, jestliže $a_{n+1} > a_n$ pro všechna $n \in \mathbb{N}$,
  \item \emph{nerostoucí}, jestliže $a_{n+1} \leq a_n$ pro všechna $n \in \mathbb{N}$,
  \item \emph{klesající}, jestliže $a_{n+1} < a_n$ pro všechna $n \in \mathbb{N}$,
  \item \emph{neklesající}, jestliže $a_{n+1} \geq a_n$ pro všechna $n \in \mathbb{N}$.
\end{itemize}
\end{definition}

\begin{definition}
Posloupnost $\{a_n\}$ je \emph{omezená}, jestliže existuje reálné číslo $M > 0$ tak, že
\[
  |a_n| \leq M \quad \text{pro všechna } n \in \mathbb{N}.
\]
Ekvivalentně, posloupnost je omezená shora, pokud $\exists K \in \mathbb{R}$ tak, že $a_n \leq K$ pro všechna $n$;  
omezená zdola, pokud $\exists L \in \mathbb{R}$ tak, že $a_n \geq L$ pro všechna $n$.
\end{definition}


\begin{definition}
Pro posloupnost $\{a_n\}$ definujeme:
\begin{itemize}
  \item \emph{maximum} $\max\{a_n\}$, pokud existuje člen $a_k$, pro který platí $a_k \geq a_n$ pro všechna $n$,
  \item \emph{minimum} $\min\{a_n\}$, pokud existuje člen $a_k$, pro který platí $a_k \leq a_n$ pro všechna $n$.
\end{itemize}
\end{definition}

První přiblížení („definice“):
Algoritmus je posloupnost instrukcí pro řešení problému.

\section{Základní prvky pseudokódu (podle Cormen et al.)}

Pseudokód se používá pro vyjádření algoritmů nezávisle na konkrétním programovacím jazyce. Níže jsou základní prvky, které se běžně používají.

\begin{pseudocodecard}{Přiřazení}
\[
x \gets y
\]
\end{pseudocodecard}

\begin{pseudocodecard}{Podmínka}
\begin{algorithmic}[1]
  \If{podmínka}
    \State příkazy
  \Else
    \State příkazy
  \EndIf
\end{algorithmic}
\end{pseudocodecard}

\begin{pseudocodecard}{Cyklus \texttt{for}}
\begin{algorithmic}[1]
  \For{$i \gets 1 \textbf{ to } n$}
    \State příkazy
  \EndFor
\end{algorithmic}
\end{pseudocodecard}

\begin{pseudocodecard}{Cyklus \texttt{while}}
\begin{algorithmic}[1]
  \While{podmínka}
    \State příkazy

\end{algorithmic}
\end{pseudocodecard}

\begin{algorithmic}[1]

  \Until{podmínka}

\end{pseudocodecard}

\section{Ukázkový algoritmus: Lineární vyhledávání}

Lineární vyhledávání prochází pole po jednotlivých prvcích a vrací index hledané hodnoty, pokud existuje.

\begin{algorithm}
\caption{LINEAR-SEARCH$(A,v)$}
\begin{algorithmic}[1]
  \For{$i \gets 1 \ \textbf{to} \ \text{length}[A]$}
    \If{$A[i] = v$}
      \State \Return $i$
    \EndIf
  \EndFor
  \State \Return NIL
\end{algorithmic}
\end{algorithm}


\section{Úkoly}
\begin{enumerate}
	\item Navrhněte v pseudokódu algoritmus který:
		\begin{itemize}
			\item IN: číslo $x$, OUT: jeho ciferný součet
			\item IN: číslo $x$, OUT: součet jeho dělitelů
			\item IN: číslo $x$, OUT: rozhodne jestli je prvočíslem nebo ne
			\item IN: číslo $x$, OUT: počet jeho dělitelů
			\item IN: pole čísel $A$, OUT: pole ve kterém jsou prvky v opačném pořadí
			\item IN: pole čísel $A$, OUT: nejmenší a největší prvek v poli v jednom průchodu
			\item IN: pole čísel $A$, OUT: součet všech prvků
			\item IN: pole čísel $A$, OUT: počet čísel větších než je průměr pole
			\item IN: pole čísel $A$, OUT: pole $A$ které neobsahuje duplicity
			\item IN: pole čísel $A$, OUT: pole kde všechny 0 jsou posunuty na konec, jinak je pořadí ponecháno
			\item IN: pole čísel $A$ a hledané číslo $x$, OUT: index prvního výskytu čísla $x$
			\item IN: pole čísel $A$ a hledané číslo $x$, OUT: pole indexů, na kterých se nachází číslo $x$
		\end{itemize}
		
\end{enumerate}

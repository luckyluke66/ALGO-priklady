\chapter{Posloupnosti}
\begin{definition}
\emph{Posloupnost} je zobrazení z množiny přirozených čísel do libovolné množiny.
Zápis: $(a_n)$ -- posloupnost, $a_n$ -- n-tý prvek.
\end{definition}

Často za množinu $A$ budeme dosazovat $\mathbb{R}$, takovým posloupnostem se říká \emph{číselné posloupnosti}. Ty budou důležité pro tuto kapitolu. O posloupnosti $(a_n):\mathbb{N} \rightarrow A$ můžeme uvažovat tak, že každému prvku z $A$ lze přiřadit index.

\section{Způsoby zadávání posloupností} 
	\subsection{Prvních x členů}
		Musí být jasné pravidlo, jak vyjádřit další členy posloupnosti.
		\begin{example}
			\begin{itemize}
				\item triviální příklad $1, 2, 3, 4 ... $
				\item fibbonacci $0, 1, 1, 2, 3, 5, 8 ...$
				\item alternující $1, -1, 1, -1 ...$
				\item konečná $2, 4, 6, ..., 20$
			\end{itemize}
		\end{example}
	\subsection{Vzorec pro n-tý člen}
		Vyjádříme obecný vzorec pro n-tý prvek na základě indexu.
		\begin{example}
			\begin{itemize}
				\item $(\frac{n}{n+1})$
				\item $((-1)^nn)$
				\item $ (1 + \frac{1}{n})^n)$
			\end{itemize}
		\end{example}
	
	\subsection{Rekurentně} 
		Rekurentní zadání obsahuje zpravidla 1. člen (nebo několik prvních členů) a pravidlo, jak vytvořit další člen ze členů předcházejících.
		\begin{example}
			\begin{itemize}
				\item $(\frac{n}{n+1})$
				\item $((-1)^nn)$
				\item $ (1 + \frac{1}{n})^n)$
			\end{itemize}
		\end{example}
Speciálním případem rekurentního zadání jsou aritmetická a geometrická posloupnost.

\begin{definition}
	Posloupnost $(a_n)$ se nazývá \textbf{aritmetická} právě tehdy, když
	\[
	\exists d \in \mathbb{R} \ \forall n \in \mathbb{N} \quad a_{n+1} = a_n + d
	\]
	Číslo $d$ se nazývá \textbf{diference} aritmetické posloupnosti.
\end{definition}

\begin{definition}
	Posloupnost $(a_n)$ se nazývá \textbf{geometrická} právě tehdy, když
	\[
	\exists q \in \mathbb{R} \ \forall n \in \mathbb{N} \quad a_{n+1} = a_n \cdot q
	\]
	Číslo $q$ se nazývá \textbf{kvocient} aritmetické posloupnosti.
\end{definition}
		
\section{Vlastnosti posloupností}
\begin{definition}
Posloupnost $(a_n)$ nazveme
\begin{itemize}
  \item \emph{rostoucí}, jestliže $a_{n+1} > a_n$ pro všechna $n \in \mathbb{N}$,
  \item \emph{nerostoucí}, jestliže $a_{n+1} \leq a_n$ pro všechna $n \in \mathbb{N}$,
  \item \emph{klesající}, jestliže $a_{n+1} < a_n$ pro všechna $n \in \mathbb{N}$,
  \item \emph{neklesající}, jestliže $a_{n+1} \geq a_n$ pro všechna $n \in \mathbb{N}$.
\end{itemize}
\end{definition}

\begin{definition}
Posloupnost $\{a_n\}$ je \emph{omezená}, jestliže existuje reálné číslo $M > 0$ tak, že
\[
  |a_n| \leq M \quad \text{pro všechna } n \in \mathbb{N}.
\]
Ekvivalentně, posloupnost je omezená shora, pokud $\exists K \in \mathbb{R}$ tak, že $a_n \leq K$ pro všechna $n$;  
omezená zdola, pokud $\exists L \in \mathbb{R}$ tak, že $a_n \geq L$ pro všechna $n$.
\end{definition}


\begin{definition}
Pro posloupnost $\{a_n\}$ definujeme:
\begin{itemize}
  \item \emph{maximum} $\max\{a_n\}$, pokud existuje člen $a_k$, pro který platí $a_k \geq a_n$ pro všechna $n$,
  \item \emph{minimum} $\min\{a_n\}$, pokud existuje člen $a_k$, pro který platí $a_k \leq a_n$ pro všechna $n$.
\end{itemize}
\end{definition}

\section{Základní otázky o algoritmech}

\subsection{správnost}
Vrátí algoritmus správný výsledek pro daný vstup? Řeší tedy algoritmus nějaký problém? 

\begin{algorithm}
\caption{Algoritmus pro nalezení maxima pole}
\begin{algorithmic}[1]
\Require Pole $A$ délky $n$ obsahující celá čísla
\Ensure Maximální prvek v poli
\State $max \gets 0$
\For{$i = 1$ to $n$}
    \If{$A[i] > max$}
        \State $max \gets A[i]$
    \EndIf
\EndFor
\State \Return $max$
\end{algorithmic}
\end{algorithm}

Co takový algoritmus? Je správný? 
Pokud chceme ukázat, že algoritmus správný není stačí nám najít vstup pro který algoritmus nenajde správný výstup.Zkusme tedy pole [1, 2, 3] maximum by mělo být 3. Po krokování algoritmu zjistíme, že tomu tak je. 
Co vstup [-1, -3, -5]? Algoritmus vrátí 0. Nula není největší prvek pole. Dokonce do něj vůbec nepatří.

Musíme si dávat pozor. Často se správnost algoritmu dokazuje. 
Ukažme snadný důkaz správnosti následujícího algoritmu.

\begin{algorithm}
\caption{Lineární vyhledávání}
\begin{algorithmic}[1]
\Require Pole $A$ délky $n$, hledaná hodnota $x$
\Ensure Index prvku $x$ nebo NIL, pokud se nenachází
\For{$i = 0$ to $n - 1$}
    \If{$A[i] = x$}
        \State \Return $i$
    \EndIf
\EndFor
\State \Return NIL
\end{algorithmic}
\end{algorithm}

\begin{proof}
	Chceme dokázat, že: Algoritmus vrátí správný výsledek a také že vždy skončí.
	Na začátku každé iterace s indexem i platí:
	„Prvek x není mezi A[0..i]“
	
	Před začátkem cyklu jsme neprohledali žádný prvek. Tedy A[0..0] neobsahuje hledaný prvek $x$.
	Teď ukažme, že to platí pro i-tou iteraci. 
 	
 	Pokud A[i] = x, algoritmus končí a vrátí $i$
 	
 	Pokud A[i] != x, tak platí, že po provedení iterace x není v A[0..i].
 	
 	For cyklus končí pokud i > n - 1. Tedy i náš algoritmus skončía vrátí NIL , což je správně protože nenašel $x$.
\end{proof}

\subsection{Optimalita}
Je náš algoritmus nejlepší možný vzhledem ke složitosti? 
Neexistuje lepší varianta? 

Podívejme se na algoritmus, který hledá GCD -- nejmenší společný dělitel.
\begin{algorithm}
\caption{Největší společný dělitel}
\begin{algorithmic}
\Require celá čísla m, n
\Ensure Největší společný dělitel čísel m, n
\While{$m$ mod $t$ $\neq$ 0 or $n$ mod $t$ $\neq$ 0}
	\State $t \gets t - 1$
\EndWhile
\State \Return t
\end{algorithmic}
\end{algorithm}

Správnost je triviální. Co optimalita? Neexistuje lepší algoritmus?

			\item IN: pole čísel $A$ a hledané číslo $x$, OUT: pole indexů, na kterých se nachází číslo $x$
		\end{itemize}
		
\end{enumerate}


\chapter{Složitost}

Analýza algoritmu - snažíme se odhadnout jak se algoritmus bude chovat. 
Pokud mluvíme o složitosti jde nám o to kolik algoritmu zabere času než nám dá výsledek.
Nebo můžeme uvažovat paměťovou složitost -- kolik paměti algoritmus bude potřebovat na výpočet.

pokud budeme dále mluvit o složitosti algoritmu. Budeme tím myslet časovou, pokud nebude explicitně řečeno jinak. 

\begin{definition}
	\emph{Časová složitost algoritmu} $A$ je funkce $T_A: \mathbb{N} \rightarrow \mathbb{N}$
	$T_A(n) =$ počet elementárních kroků, které $A$ vykoná pro vstup velikosti n.
\end{definition}

Funkce $T_A(n)$ popisuje jak se algoritmus chová pro všechny vstupy.
Obvykle nás zajímá nejhorší případ, nebo průměrný případ.


Nechť algoritmus $A$ řeší problém $P$ a nechť $I_1, I_2, \dots, I_m$ jsou všechny možné vstupy problému $P$ o velikosti $n$.  
Označme $t_A(I_i)$ dobu běhu algoritmu $A$ na instanci $I_i$.

\begin{definition}
\textbf{Časová složitost v nejhorším případě} (worst-case time complexity) je definována jako maximální doba běhu algoritmu pro všechny možné instance o velikosti $n$:
\[
T_A^{\max}(n) = max\{t_A(I)|\text{I je vstup problému P a} |I| = n\}
\]

\[
\qquad \text{tj. graficky:} \qquad
\left.
\begin{array}{c|c}
I_1 & t_A(I_1) \\
I_2 & t_A(I_2) \\
\vdots & \vdots \\
I_m & t_A(I_m)
\end{array}
\right\}
\text{max.}
\]
\end{definition}
\newpage
\begin{definition}
\textbf{Časová složitost v průměrném případě} (average-case time complexity) je definována jako průměrná doba běhu algoritmu:
\[
T_A^{\text{avg}}(n) = \frac{t_A(I_1) + \dots + t_A(I_m)}{m}
\qquad \text{tj. graficky:} \qquad
\left.
\begin{array}{c|c}
I_1 & t_A(I_1) \\
I_2 & t_A(I_2) \\
\vdots & \vdots \\
I_m & t_A(I_m)
\end{array}
\right\}
\mathbb{E}
\]
kde $p_i$ je pravděpodobnost výskytu instance $I_i$.
\end{definition}

\Function{Faktoriál}{$n$}

    \EndIf

\begin{algorithm}[H]
\caption{Eukleidův algoritmus}
\begin{algorithmic}
\Function{NSD}{$a, b$}
    \While{$b \neq 0$}
        \State $r \gets a \bmod b$
        \State $a \gets b$
        \State $b \gets r$
    \EndWhile
    \State \Return $a$
\EndFunction
\end{algorithmic}
\end{algorithm}


\begin{algorithm}[H]
\caption{Výpočet $n$-tého Fibonacciho čísla}
\begin{algorithmic}
\Function{Fibonacci}{$n$}
    \If{$n \leq 1$}
        \State \Return $n$
    \EndIf
    \State $prev \gets 0$
    \State $curr \gets 1$
    \For{$i \gets 2$ \textbf{to} $n$}
        \State $new \gets prev + curr$
        \State $prev \gets curr$
        \State $curr \gets new$
    \EndFor
    \State \Return $curr$
\EndFunction
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Rychlé mocnění}
\begin{algorithmic}
\Function{Mocnina}{$x, n$}
    \If{$n = 0$}
        \State \Return $1$
    \ElsIf{$n$ je sudé}
        \State $y \gets$ \Call{Mocnina}{$x, n/2$}
        \State \Return $y \times y$
    \Else
        \State \Return $x \times$ \Call{Mocnina}{$x, n-1$}
    \EndIf
\EndFunction
\end{algorithmic}
\end{algorithm}

\chapter{Třídění}

\begin{definition}
\emph{Problém třídění:}

\textbf{Vstup:} posloupnost n čísel: $\langle a_1, a_2 \dots a_n \rangle$

\textbf{Výstup:} permutace $\langle b_1, b_2 \dots b_n \rangle$ tak, že platí 
$b_1 \leq b_2 \leq \dots \leq b_n$

vstup i výstup obvykle reprezentujeme polem.
\end{definition}

\begin{definition}

Algoritmus pracuje \emph{na místě} pokud použije nanejvýš konstantní prostor mimo tříděné pole.  

\end{definition}


\section{Insertion Sort}

Idea tohoto algoritmu je podobná způsobu, jak třídíme \( n \) rozdaných karet:  
\( n \) karet leží na začátku na stole. Pravou rukou je bereme a vkládáme do levé ruky tak, že v levé ruce vzniká setříděná posloupnost karet (zleva od nejmenší po největší). Drží-li levá ruka \( k \) karet, pak další, \((k + 1)\)-ní, kartu zatřídíme tak, že ji zprava porovnáváme se setříděnými kartami a vložíme ji na správné místo.

\begin{algorithm}[h!]
\caption{Insertion-Sort(\(A[0..n-1]\))}
\begin{algorithmic}[1]
\For{$j \gets 1$ \textbf{to} $n - 1$}
  \State $t \gets A[j]$
  \State $i \gets j - 1$
  \While{$i \ge 0$ \textbf{and} $A[i] > t$}
    \State $A[i + 1] \gets A[i]$
    \State $i \gets i - 1$
  \EndWhile
  \State $A[i + 1] \gets t$
\EndFor
\end{algorithmic}
\end{algorithm}

\section{Selection Sort}

Idea: Najdi v \( A[0..n-1] \) nejmenší prvek a vyměň ho s \( A[0] \).  
Najdi v \( A[1..n-1] \) nejmenší prvek a vyměň ho s \( A[1] \).  
\(\ldots\)  
Najdi v \( A[n-2..n-1] \) nejmenší prvek a vyměň ho s \( A[n-2] \).

\begin{algorithm}[h!]
\caption{Selection-Sort(\(A[0..n-1]\))}
\begin{algorithmic}[1]
\For{$j \gets 0$ \textbf{to} $n - 2$}
  \State $iMin \gets j$
  \For{$i \gets j + 1$ \textbf{to} $n - 1$}
    \If{$A[i] < A[iMin]$}
      \State $iMin \gets i$
    \EndIf
  \EndFor
  \State $t \gets A[j]; \; A[j] \gets A[iMin]; \; A[iMin] \gets t$
\EndFor
\end{algorithmic}
\end{algorithm}

		\item Upravte insertion sort tak, aby třídíl sestupně nebo vzestupně na základě volitelného parametru.
	\end{enumerate}


\chapter{Asymptotický růst funkcí}

V předchožích kapitolách jsme počítaly kolik kroků algoritmus vykoná v nejhorším případě (časová složistos). V této kapitole se budeme věnovat časové složitosti více do hloubky. Zavedeme $O$-notaci.
Ukážeme jak analyzovat algoritmy efektivněji bez zbytečně velké přesnosti, která nám nedává informaci navíc. 
Dejme tomu, že algoritmus $A$ a $B$ řeší problém $P$. Spočítáním počtu instrukcí co provedou máme: 
$T_A(n) = 2n + 4$ a $T_B(n) = 6n - 10$. 

Vidíme, že funkce $T_A(n)$ roste rychleji než $T_B(n)$ rozdíl je ale nezávislý na n.
Podívejme se tedy na hrubší metodu analýzy, která tento nedostatek odstraní.

\begin{definition}
\emph{Asymptotická hodní mez $O(g)$} pro funkci $g(n)$ je 
\[
O(g(n)) = \{f(n) \ | \ \exists c > 0 \ \text{a} \ n_0 \in \mathbb{N} \text{ tak, že } \forall n \geq n_0 \text{ je } 0 \leq f(n) \leq cg(n))\}
\]
\end{definition}

Všimněte, že definujeme množinu funkcí, které splňují podmínku "$f$ funkce je zhora omezená funkcí $g$".
Pak můžeme říct $n^3 + 2n + 4 \in O(n^3)$. Tedy že funkce$n^3 + 2n + 4$ roste nanejvýš tak rychle jako 
$c(n^3)$. Bude se nám hodit ještě pár definic. 

\begin{definition}
\emph{Asymptotická dolní mez $\Omega(g)$} Pro funkci $g(n)$ je
\[
\Omega(g(n)) = \{ f(n) \ | \ \exists c > 0,\ n_0 \in \mathbb{N} \forall n \ge n_0 \text{ je } 0 \le c g(n) \le f(n) \}.
\]
\end{definition}

    \caption{Znázornění funkce $f(n)$ mezi $c_1 g(n)$ a $c_2 g(n)$.}
    \label{fig:theta}
\end{figure}



\begin{definition}
\emph{Asymptotická oboustranná (těsná) mez} Pro funkci $g(n)$ je
\[
\Theta(g(n)) = \{ f(n) \mid \exists\, c_1, c_2 > 0,\ n_0 \in \mathbb{N} \forall n \ge n_0 \text{ je } 0 \le c_1 g(n) \le f(n) \le c_2 g(n) \}.
\]

Alternativně platí, že $f(n) \in \Theta(g(n))$ právě když $f(n) \in O(g(n))$ a $f(n) \in \Omega(g(n))$
\end{definition}

\section{Vlastnosti}

\begin{pseudocodecardorange}{Tranzitivita odhadů}
Pokud $f = O(g)$ a $g = O(h)$, pak $f = O(h)$.\\
Pokud $f = \Omega(g)$ a $g = \Omega(h)$, pak $f = \Omega(h)$.\\
Pokud $f = \Theta(g)$ a $g = \Theta(h)$, pak $f = \Theta(h)$.\\
Pokud $f = o(g)$ a $g = o(h)$, pak $f = o(h)$.\\
Pokud $f = \omega(g)$ a $g = \omega(h)$, pak $f = \omega(h)$.
\end{pseudocodecardorange}

\begin{pseudocodecardorange}{Reflexitita a symetrie odhadů}
Reflexivita:\\
$f = O(f).$\\
$f = \Omega(f).$\\
$f = \Theta(f).$\\
Symetrie:\\
$f = \Theta(g)$ právě když $g = \Theta(f)$.\\
$f = O(g)$ právě když $g = \Omega(f)$.\\
$f = o(g)$ právě když $g = \omega(f)$.\\
\end{pseudocodecardorange}

	\item Rozhodněte, zda platí následující: 
		 \begin{itemize}
		 	\item $3n^2 + 5n + 1 = \Theta(n^2)$
		 	\item $2n^3 - 10 = O(n^3)$
		 	\item $12 \log{n} + 3n = \Theta(n \log n)$

	\item Dokažte alternativní definici $\Theta$-notace.
	 
\end{enumerate}
